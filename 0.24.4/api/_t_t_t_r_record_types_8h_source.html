71// Becker Hickl SPC-600/630 4096 Channel Mode.
72// The information about the subsequent photons is stored one after another in the measurement
73// data file. For each photon 6 bytes are used. The parameter @param adc corresponds to the value of
74// the analog to digital converter and related to the micro time (mt) by
75// mt = (4095 - adc) * TACRange / (TACGain * 4096)
76// @param invalid is true for invalid TTTR records. If invalid is true all data in the record
77// except the mtov bit are invalid. @param mtov marks that a macro timer overflows. In case @param mtov
78// is true 2**24 counts should be added to the overflow counter (the overflow counter counts the
79// number of overflows. @param gap marks possible gaps, e.g., due to FIFO overflows. @param rout
80// provides the (inverted) routing number of the TTTR record.
81typedef union bh_spc600_4096_record{
82 uint32_t allbits;

85 unsigned invalid :1;
86 unsigned mtov :1;
87 unsigned gap :1;
88 unsigned empty :1;
89 unsigned mt3 :8;

93 } bits;
94} bh_spc600_4096_record_t;
